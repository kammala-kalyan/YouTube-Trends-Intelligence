1. Know the data limits first
State and city data comes only from Google Trends. The YouTube Data API doesn’t expose viewer geography for public videos. Use Trends with gprop=youtube (YouTube Search) for regional interest by state, and by city for some queries.
Trends is relative, not absolute. It returns an index from 0 to 100 per query batch, so you can compare topics only within the same request (max 5 terms). Use a shared anchor keyword across batches to normalize.
Google Trends has no stable public API. pytrends is unofficial and rate-limits quickly, so add sleeps and retries and cache everything. Check whether Google’s newer official Trends API access is available to you.
YouTube API quota is 10,000 units/day. videos.list costs 1 unit, search.list costs 100. Prefer videos.list with chart=mostPopular&regionCode=IN&videoCategoryId=..., and use search.list sparingly.
Regional language opportunities come from the video’s defaultAudioLanguage/snippet.defaultLanguage (often missing) and from running Trends queries in Hindi, Tamil, Telugu, Kannada, etc. Add a title-script detection step (Devanagari, Tamil script, and so on) to fill the gaps.

6. Key DAX measures
DAX
Engagement Rate = DIVIDE(SUM(fact[likes]) + SUM(fact[comments]), SUM(fact[views]))

Views per Day =
DIVIDE(MAX(fact[views]), DATEDIFF(MAX(dim_video[published_at]), MAX(fact[snapshot_date]), DAY) + 1)

Interest MoM % =
VAR cur = CALCULATE(AVERAGE(trends[interest]), DATESINPERIOD(dim_date[Date], MAX(dim_date[Date]), -1, MONTH))
VAR prev = CALCULATE(AVERAGE(trends[interest]), DATEADD(DATESINPERIOD(dim_date[Date], MAX(dim_date[Date]), -1, MONTH), -1, MONTH))
RETURN DIVIDE(cur - prev, prev)

Trend Status = SWITCH(TRUE(), [Interest MoM %] > 0.15, "Rising", [Interest MoM %] < -0.15, "Declining", "Stable")
Content gap score idea: search interest (normalized) ÷ number of recent videos on that topic. High means demand exceeds supply. Compute the video count by sampling search.list for a small, curated keyword list to protect quota.

import os, datetime as dt
import requests, pandas as pd
from config import API_KEY, REGION_CODE, DATA_DIR

BASE = "https://www.googleapis.com/youtube/v3"
TODAY = dt.date.today().isoformat()
os.makedirs(DATA_DIR, exist_ok=True)

def get_categories():
    r = requests.get(f"{BASE}/videoCategories",
                     params={"part": "snippet", "regionCode": REGION_CODE, "key": API_KEY}).json()
    rows = [dict(category_id=i["id"], category_name=i["snippet"]["title"],
                 assignable=i["snippet"].get("assignable", False))
            for i in r.get("items", [])]
    return pd.DataFrame(rows)

def get_trending(category_id=None, max_pages=4):
    rows, token, rank = [], None, 0
    for _ in range(max_pages):
        params = dict(part="snippet,statistics,contentDetails", chart="mostPopular",
                      regionCode=REGION_CODE, maxResults=50, key=API_KEY)
        if category_id:
            params["videoCategoryId"] = category_id
        if token:
            params["pageToken"] = token
        r = requests.get(f"{BASE}/videos", params=params)
        if not r.ok:   # some categories have no trending chart
            print(f"  skipped category {category_id}: {r.status_code}")
            return rows
        data = r.json()
        for v in data.get("items", []):
            rank += 1
            s, st, cd = v["snippet"], v.get("statistics", {}), v["contentDetails"]
            rows.append(dict(
                snapshot_date=TODAY, category_chart=category_id or "overall", rank=rank,
                video_id=v["id"], title=s["title"], channel_id=s["channelId"],
                channel_title=s["channelTitle"], category_id=s["categoryId"],
                published_at=s["publishedAt"], duration=cd["duration"],
                audio_lang=s.get("defaultAudioLanguage"), default_lang=s.get("defaultLanguage"),
                tags="|".join(s.get("tags", [])),
                views=int(st["viewCount"]) if "viewCount" in st else None,
                likes=int(st["likeCount"]) if "likeCount" in st else None,
                comments=int(st["commentCount"]) if "commentCount" in st else None))
        token = data.get("nextPageToken")
        if not token:
            break
    return rows

if __name__ == "__main__":
    cats = get_categories()
    cats.to_csv(f"{DATA_DIR}/dim_category.csv", index=False)
    print(f"Saved {len(cats)} categories")

    all_rows = get_trending()                      # overall India chart
    for cid in cats[cats.assignable]["category_id"]:
        print(f"Category {cid}")
        all_rows += get_trending(cid)

    df = pd.DataFrame(all_rows)
    df.to_csv(f"{DATA_DIR}/trending_videos_{TODAY}.csv", index=False)
    print(f"Saved {len(df)} rows, {df.video_id.nunique()} unique videos, "
          f"{df.channel_id.nunique()} unique channels")

Script 4: Google Trends (over time and by state)
This uses no YouTube quota. Create fetch_trends.py:

python
import time, datetime as dt
import pandas as pd
from pytrends.request import TrendReq
from config import DATA_DIR
from keywords import KEYWORDS, ANCHOR

TODAY = dt.date.today().isoformat()
pt = TrendReq(hl="en-IN", tz=330, retries=3, backoff_factor=1.5)

rows = [(cat, kw) for cat, items in KEYWORDS.items() for kw, _ in items if kw != ANCHOR]
cat_of = {kw: cat for cat, kw in rows}
kws = list(cat_of)
batches = [kws[i:i + 4] for i in range(0, len(kws), 4)]

def with_retry(fn, tries=4):
    for n in range(tries):
        try:
            return fn()
        except Exception as e:
            wait = 30 * (n + 1)
            print(f"  retry in {wait}s ({e})")
            time.sleep(wait)
    return None

over_time, by_state = [], []
for b_id, batch in enumerate(batches):
    terms = batch + [ANCHOR]
    print(f"Batch {b_id + 1}/{len(batches)}: {terms}")
    pt.build_payload(terms, geo="IN", timeframe="today 5-y", gprop="youtube")

    ot = with_retry(pt.interest_over_time)
    if ot is not None and not ot.empty:
        ot = ot.drop(columns=["isPartial"], errors="ignore").reset_index()
        ot = ot.melt(id_vars="date", var_name="keyword", value_name="interest")
        ot["batch_id"] = b_id
        over_time.append(ot)
    time.sleep(8)

    bs = with_retry(lambda: pt.interest_by_region(resolution="REGION", inc_low_vol=True))
    if bs is not None and not bs.empty:
        bs = bs.reset_index().melt(id_vars="geoName", var_name="keyword", value_name="interest")
        bs = bs.rename(columns={"geoName": "state"})
        bs["batch_id"] = b_id
        by_state.append(bs)
    time.sleep(8)

ot = pd.concat(over_time)
ot["category"] = ot["keyword"].map(cat_of)
ot["snapshot_date"] = TODAY
ot.to_csv(f"{DATA_DIR}/trends_over_time_{TODAY}.csv", index=False)

bs = pd.concat(by_state)
bs["category"] = bs["keyword"].map(cat_of)
bs["snapshot_date"] = TODAY
bs.to_csv(f"{DATA_DIR}/trends_by_state_{TODAY}.csv", index=False)
print(f"Saved {len(ot)} time rows and {len(bs)} state rows")

Script 5: topic videos and content-gap supply
This uses the YouTube quota: one search call per keyword at 100 units each, so about 3,000 units for this list, which stays under your 9,000 guard. Create fetch_topics.py:

python
import datetime as dt
import requests, pandas as pd
from config import API_KEY, REGION_CODE, DATA_DIR, DAILY_QUOTA_LIMIT
from keywords import KEYWORDS

BASE = "https://www.googleapis.com/youtube/v3"
TODAY = dt.date.today().isoformat()
since = (dt.datetime.utcnow() - dt.timedelta(days=30)).strftime("%Y-%m-%dT%H:%M:%SZ")

units, hits, supply = 0, [], []
for cat, items in KEYWORDS.items():
    for kw, lang in items:
        if units + 100 > DAILY_QUOTA_LIMIT:
            print("Quota guard reached, stopping"); break
        r = requests.get(f"{BASE}/search", params=dict(
            part="snippet", q=kw, type="video", regionCode=REGION_CODE,
            relevanceLanguage=lang, order="viewCount", publishedAfter=since,
            maxResults=50, key=API_KEY))
        units += 100
        if not r.ok:
            print(f"  {kw}: error {r.status_code}"); continue
        d = r.json()
        supply.append(dict(snapshot_date=TODAY, category=cat, keyword=kw, lang=lang,
                           videos_last_30d_est=d["pageInfo"]["totalResults"]))
        hits += [dict(keyword=kw, category=cat, lang=lang, video_id=i["id"]["videoId"])
                 for i in d.get("items", [])]
        print(f"{cat} / {kw}: {len(d.get('items', []))} videos")

pd.DataFrame(supply).to_csv(f"{DATA_DIR}/keyword_supply_{TODAY}.csv", index=False)

# fetch full stats for the videos found (1 unit per 50 videos)
hit_df = pd.DataFrame(hits).drop_duplicates(["keyword", "video_id"])
ids = hit_df["video_id"].unique().tolist()
rows = []
for i in range(0, len(ids), 50):
    r = requests.get(f"{BASE}/videos", params=dict(
        part="snippet,statistics,contentDetails", id=",".join(ids[i:i + 50]), key=API_KEY))
    units += 1
    for v in r.json().get("items", []):
        s, st, cd = v["snippet"], v.get("statistics", {}), v["contentDetails"]
        rows.append(dict(
            video_id=v["id"], title=s["title"], channel_id=s["channelId"],
            channel_title=s["channelTitle"], category_id=s["categoryId"],
            published_at=s["publishedAt"], duration=cd["duration"],
            audio_lang=s.get("defaultAudioLanguage"), default_lang=s.get("defaultLanguage"),
            tags="|".join(s.get("tags", [])),
            views=int(st["viewCount"]) if "viewCount" in st else None,
            likes=int(st["likeCount"]) if "likeCount" in st else None,
            comments=int(st["commentCount"]) if "commentCount" in st else None))

stats = pd.DataFrame(rows)
out = hit_df.merge(stats, on="video_id", how="left")
out["snapshot_date"] = TODAY
out.to_csv(f"{DATA_DIR}/topic_videos_{TODAY}.csv", index=False)
print(f"Saved {len(out)} rows, {out.video_id.nunique()} unique videos. Quota used: ~{units}")
